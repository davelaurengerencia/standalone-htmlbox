// Resuelve URLs relativas a absolutas para evitar errores en entornos blob: o iframe
function getApiUrl(endpoint) {
  if (!endpoint) return '';
  if (endpoint.startsWith('http://') || endpoint.startsWith('https://')) {
    return endpoint;
  }
  const base = (window.EDITOR_BASE || '').replace(/\/$/, '');
  const path = endpoint.startsWith('/') ? endpoint : '/' + endpoint;
  if (base) {
    return base + path;
  }
  if (window.location && window.location.origin && window.location.origin !== 'null' && !window.location.protocol.startsWith('blob:')) {
    return window.location.origin + path;
  }
  return 'http://localhost' + path;
}

// Wrapper seguro para peticiones fetch con soporte para backend local / preview independiente
// El fallback a datos simulados es OPT-IN, y esto es importante: antes se
// disparaba solo, ante CUALQUIER error de red. Servido en producción eso
// significa que si la API está caída, el editor muestra un catálogo de nodos
// inventado y el usuario "guarda" un flujo que no se guardó en ningún lado —
// con un warning en la consola como única señal. Un error de red tiene que
// verse como un error, no maquillarse.
//
// Para trabajar en el editor sin backend (preview, diseño de la UI), poner
// `window.EDITOR_MOCK = true` antes de que cargue este script.
async function apiFetch(endpoint, options = {}) {
  const url = getApiUrl(endpoint);
  if (window.EDITOR_MOCK === true) {
    return await mockApiHandler(endpoint, options);
  }
  const res = await fetch(url, options);
  if (!res.ok) {
    throw new Error(`La API del editor respondió HTTP ${res.status} en ${endpoint}`);
  }
  return await res.json();
}

// Fallback intercepción API cuando fetch de red falla o corre en preview sin backend real
async function mockApiHandler(endpoint, options = {}) {
  const method = (options.method || 'GET').toUpperCase();
  const body = options.body ? JSON.parse(options.body) : null;
  const [path, queryString] = endpoint.split('?');
  const params = new URLSearchParams(queryString || '');

  if (path === '/_editor/api/nodes') return DEFAULT_NODE_DEFS;
  if (path === '/_editor/api/flows') return Object.keys(localStore.flows);
  
  if (path.startsWith('/_editor/api/flows/')) {
    const name = decodeURIComponent(path.replace('/_editor/api/flows/', ''));
    if (method === 'GET') return localStore.flows[name] || [];
    if (method === 'POST') {
      localStore.flows[name] = body;
      return { ok: true };
    }
  }

  if (path === '/_editor/api/context') return localStore.context;
  if (path === '/_editor/api/context/global' && method === 'POST') {
    localStore.context.global = body.value || {};
    return { ok: true };
  }
  if (path.startsWith('/_editor/api/context/flow/') && method === 'POST') {
    const name = decodeURIComponent(path.replace('/_editor/api/context/flow/', ''));
    localStore.context.flows[name] = body.value || {};
    return { ok: true };
  }

  if (path === '/_editor/api/config-nodes') {
    if (method === 'GET') {
      const typeFilter = params.get('type');
      if (typeFilter) {
        return localStore.configNodes.filter(c => c.type === typeFilter);
      }
      return localStore.configNodes;
    }
    if (method === 'POST') {
      if (body.id) {
        const idx = localStore.configNodes.findIndex(c => c.id === body.id);
        if (idx !== -1) localStore.configNodes[idx] = { ...localStore.configNodes[idx], ...body };
      } else {
        body.id = 'cfg_' + Date.now().toString(36);
        body.hasToken = Boolean(body.token);
        body.tokenPreview = body.token ? '••••' + body.token.slice(-4) : '';
        localStore.configNodes.push(body);
      }
      return { ok: true };
    }
  }

  if (path.startsWith('/_editor/api/config-nodes/')) {
    const parts = path.split('/');
    const connId = parts[4];
    if (parts.length === 5 && method === 'DELETE') {
      localStore.configNodes = localStore.configNodes.filter(c => c.id !== connId);
      return { ok: true };
    }
    if (path.endsWith('/migrations')) {
      if (method === 'GET') return { migrations: localStore.migrations[connId] || [], applied: [] };
      if (method === 'POST') {
        localStore.migrations[connId] = localStore.migrations[connId] || [];
        const version = localStore.migrations[connId].length + 1;
        localStore.migrations[connId].push({ version, ...body });
        return { ok: true };
      }
    }
    if (path.endsWith('/migrations/apply')) return { results: [{ ok: true, version: 1 }] };
    if (path.endsWith('/flexdb-sheets')) return { sheets: [] };
  }

  if (path === '/_editor/api/run') {
    const flow = body.flow || [];
    const logs = [`⚡ Ejecutando flujo local "${body.flowName || 'demo'}" (${flow.length} nodos)...`];
    let payload = { demo: true, timestamp: Date.now() };
    flow.forEach(node => {
      if (node.type === 'function' && node.code) {
        try {
          const fn = new Function('msg', 'ctx', node.code);
          const res = fn({ payload }, { env: {}, log: (...a) => logs.push(a.join(' ')) });
          if (res && res.payload !== undefined) payload = res.payload;
        } catch (e) {
          logs.push(`⚠ Error en función "${node.name || node.id}": ${e.message}`);
        }
      }
    });
    return { ok: true, logs, result: { status: 'success', payload } };
  }

  if (path === '/_editor/api/run-node') {
    return { result: { tested: true, nodeType: body.type, payload: body.payload } };
  }

  return { ok: true };
}

function uid(prefix) {
  if (prefix === 'n') {
    if (globalThis.crypto?.randomUUID) {
      return globalThis.crypto.randomUUID().replace(/-/g, '').slice(0, 16);
    }
    return `${Date.now().toString(36)}${Math.random().toString(36).slice(2, 10)}`;
  }
  return prefix + '_' + (idCounter++);
}

function resyncIdCounterFromLoadedIds(ids) {
  let maxSeen = 0;
  (ids || []).forEach(id => {
    const m = String(id || '').match(/_(\d+)$/);
    if (m) maxSeen = Math.max(maxSeen, parseInt(m[1], 10));
  });
  if (maxSeen + 1 > idCounter) idCounter = maxSeen + 1;
}

function outputCountFor(node, def) {
  if (node.type === 'switch') {

      const rules = JSON.parse(node.config.rules || '[]');
      return Array.isArray(rules) && rules.length ? rules.length : 1;
    } catch { return 1; }
  }
  return def.outputs ?? 1;
}

const DEBUG_LINE_RE = /^🐛 \[(.*?)\] (.+)$/
function logLine(msg, isErr) {
  const el = document.getElementById('console');
  const entry = document.createElement('div');
  entry.className = 'entry';

  const time = document.createElement('div');
  time.className = 'entry-time';

  line.className = 'line' + (isErr ? ' err' : '');

    let pretty = m[2];

function renderPalette() {
  const nt = document.getElementById('node-types');
  nt.innerHTML = '';
  const categories = [...new Set(NODE_DEFS.map(d => d.category))];
  categories.forEach(cat => {
    const h = document.createElement('h3');
    h.textContent = cat;
    nt.appendChild(h);
    NODE_DEFS.filter(d => d.category === cat).forEach(def => {
      const el = document.createElement('div');
      el.className = 'palette-item';
      el.style.borderLeftColor = colorFor(def.type);
      const warn = def.type === 'function'
        ? ' <span title="⚠ Ejecuta código SIN sandbox">⚠</span>'
        : '';
      el.innerHTML = def.name + warn + '<small>' + def.type + '</small>';
      el.draggable = true;
      el.ondragstart = e => e.dataTransfer.setData('text/plain', JSON.stringify({ type: def.type }));
      nt.appendChild(el);
    });
  });
}

function colorFor(type) {
  const palette = ['#7aa2f7', '#bb9af7', '#7dcfff', '#73daca', '#b4f9f8', '#9ece6a', '#e0af68', '#f7768e'];
  let hash = 0;
  for (let i = 0; i < type.length; i++) hash = (hash * 31 + type.charCodeAt(i)) >>> 0;
  return palette[hash % palette.length];
}

function addNode(type, x, y, presetConfig) {
  const def = defOf(type);
  if (!def) { logLine('❌ tipo de nodo desconocido: ' + type, true); return null; }
  const config = {};
  Object.entries(def.defaults || {}).forEach(([key, cfg]) => {
    config[key] = presetConfig && presetConfig[key] !== undefined ? presetConfig[key] : cfg.value;
  });
  const node = {
    id: (presetConfig && presetConfig.__id) || uid('n'),
    type,
    name: (presetConfig && presetConfig.__name) || '',
    position: { x: Math.max(0, x), y: Math.max(0, y) },
    config,
    notes: (presetConfig && presetConfig.__notes) || '',
  };
  nodes.push(node);
  selectedNodeId = node.id;
  renderCanvas();
  renderConfig();
  return node;
}

function deleteNode(id) {
  nodes = nodes.filter(n => n.id !== id);
  edges = edges.filter(e => e.from !== id && e.to !== id);
  if (selectedNodeId === id) selectedNodeId = null;
  renderCanvas();
  renderConfig();
}

function renderCanvas() {
  const wrap = document.getElementById('canvas-wrap');
  const canvas = document.getElementById('canvas');
  canvas.querySelectorAll('.node').forEach(n => n.remove());

  wrap.ondragover = e => e.preventDefault();
  wrap.ondrop = e => {
    e.preventDefault();
    const data = JSON.parse(e.dataTransfer.getData('text/plain') || '{}');
    if (!data.type) return;
    const rect = canvas.getBoundingClientRect();
    const x = e.clientX - rect.left - 70;
    const y = e.clientY - rect.top - 20;
    addNode(data.type, x, y);
    if (data.type === 'function') {
      logLine('⚠ El nodo "function" ejecuta código SIN sandbox (new Function).', true);
    }
  };

  nodes.forEach(node => {
    const def = defOf(node.type) || { name: node.type, inputs: 1, outputs: 1 };
    const color = colorFor(node.type);
    const el = document.createElement('div');
    el.className = 'node' + (node.id === selectedNodeId ? ' selected' : '');
    el.id = 'node-' + node.id;
    el.style.left = node.position.x + 'px';
    el.style.top = node.position.y + 'px';
    el.style.borderLeftColor = color;

    let portsHtml = '';
    if ((def.inputs ?? 1) > 0) {
      portsHtml += '<div class="port in" data-node="' + node.id + '" data-role="in"></div>';
    }
    const outputs = outputCountFor(node, def);
    const PORT_GAP_PX = 28;
    if (outputs > 1) {
      const neededHeight = outputs * PORT_GAP_PX + 14;
      el.style.minHeight = neededHeight + 'px';
    }
    for (let i = 0; i < outputs; i++) {
      let topStyle;
      if (outputs === 1) {
        topStyle = 'top:50%';
      } else {
        const totalSpan = (outputs - 1) * PORT_GAP_PX;
        const offset = -totalSpan / 2 + i * PORT_GAP_PX;
        topStyle = 'top:calc(50% + ' + offset + 'px)';
      }
      portsHtml += '<div class="port out" style="' + topStyle + '" data-node="' + node.id + '" data-output="' + i + '" title="salida ' + (i + 1) + '"></div>';
      if (outputs > 1) {
        portsHtml += '<div class="port-label" style="' + topStyle + '">' + (i + 1) + '</div>';
      }
    }

    const displayName = node.name || def.name;
    const unsandboxed = node.type === 'function';
    el.title = def.name + (node.name ? ' — ' + node.name : '') + (unsandboxed ? ' — ⚠ SIN sandbox' : '') + ' (doble clic para configurar)';
    
    let entryBtn = '';
    if (node.type === 'inject' || node.type === 'http-in') {
      entryBtn = '<button class="inject-btn" title="Disparar nodo" onclick="event.stopPropagation(); triggerEntryNode(\'' + node.id + '\')">▶</button>';
    }
    const warnBadge = unsandboxed ? ' <span title="⚠ Ejecuta código SIN sandbox">⚠</span>' : '';
    el.innerHTML = '<div class="name-label">' + displayName + warnBadge + '</div>' + entryBtn + portsHtml;

    el.ondblclick = e => {
      if (e.target.classList.contains('port')) return;
      openNodeEditor(node.id);
    };

    el.onmousedown = e => {
      if (e.target.classList.contains('port') || e.target.classList.contains('inject-btn')) return;
      selectedNodeId = node.id;
      renderConfig();
      document.querySelectorAll('.node').forEach(n => n.classList.remove('selected'));
      el.classList.add('selected');

      const startX = e.clientX, startY = e.clientY;
      const ox = node.position.x, oy = node.position.y;
      function move(ev) {
        node.position.x = ox + (ev.clientX - startX);
        node.position.y = oy + (ev.clientY - startY);
        el.style.left = node.position.x + 'px';
        el.style.top = node.position.y + 'px';
        drawWires();
      }
      function up() {
        document.removeEventListener('mousemove', move);
        document.removeEventListener('mouseup', up);
      }
      document.addEventListener('mousemove', move);
      document.addEventListener('mouseup', up);
    };

    canvas.appendChild(el);

    el.querySelectorAll('.port.out').forEach(outPort => {
      outPort.onmousedown = e => {
        e.stopPropagation();
        const outputIndex = Number(outPort.dataset.output);
        const rect = canvas.getBoundingClientRect();
        wireDraft = { fromNode: node.id, fromOutput: outputIndex, x2: e.clientX - rect.left, y2: e.clientY - rect.top };
        function move(ev) {
          wireDraft.x2 = ev.clientX - rect.left;
          wireDraft.y2 = ev.clientY - rect.top;
          drawWires();
        }
        function up(ev) {
          document.removeEventListener('mousemove', move);
          document.removeEventListener('mouseup', up);
          const target = document.elementFromPoint(ev.clientX, ev.clientY);
          if (target && target.classList.contains('port') && target.dataset.role === 'in') {
            const toNode = target.dataset.node;
            if (toNode !== node.id) {
              edges.push({ id: uid('e'), from: node.id, fromOutput: outputIndex, to: toNode });
            }
          }
          wireDraft = null;
          drawWires();
        }
        document.addEventListener('mousemove', move);
        document.addEventListener('mouseup', up);
      };
    });
  });

  drawWires();
  updateDirtyIndicator();
}

function selectEdge(id) {
  selectedEdgeId = id;
  drawWires();
}

function deleteEdge(id) {
  edges = edges.filter(e => e.id !== id);
  selectedEdgeId = null;
  drawWires();
  renderConfig();
  updateDirtyIndicator();
}

function deselectEdge() {
  if (!selectedEdgeId) return;
  selectedEdgeId = null;
  drawWires();
}

function portCenter(nodeId, role, outputIndex) {
  const node = nodes.find(n => n.id === nodeId);
  if (!node) return { x: 0, y: 0 };
  const el = document.getElementById('node-' + nodeId);
  const width = el ? el.offsetWidth : 160;
  const height = el ? el.offsetHeight : 50;
  if (role === 'in') {
    return { x: node.position.x, y: node.position.y + height / 2 };
  }
  const portEl = el ? el.querySelector('.port.out[data-output="' + outputIndex + '"]') : null;
  const relY = portEl ? portEl.offsetTop + portEl.offsetHeight / 2 : height / 2;
  return { x: node.position.x + width, y: node.position.y + relY };
}

function drawWires(highlightIds) {
  const svg = document.getElementById('wires');
  svg.innerHTML = '';
  edges.forEach(edge => {
    const p1 = portCenter(edge.from, 'out', edge.fromOutput || 0);
    const p2 = portCenter(edge.to, 'in');
    const mid = (p1.x + p2.x) / 2;
    const midY = (p1.y + p2.y) / 2;
    const d = 'M ' + p1.x + ' ' + p1.y + ' C ' + mid + ' ' + p1.y + ', ' + mid + ' ' + p2.y + ', ' + p2.x + ' ' + p2.y;
    const active = highlightIds && highlightIds.includes(edge.id);
    const selected = edge.id === selectedEdgeId;

    const hit = document.createElementNS('http://www.w3.org/2000/svg', 'path');
    hit.setAttribute('d', d);
    hit.setAttribute('stroke', 'transparent');
    hit.setAttribute('stroke-width', '14');
    hit.setAttribute('fill', 'none');
    hit.style.cursor = 'pointer';
    hit.style.pointerEvents = 'stroke';
    hit.addEventListener('click', (e) => { e.stopPropagation(); selectEdge(edge.id); });
    svg.appendChild(hit);

    const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
    path.setAttribute('d', d);
    path.setAttribute('stroke', selected ? '#f7768e' : (active ? 'var(--wire-active)' : 'var(--wire)'));
    path.setAttribute('stroke-width', selected ? '3' : (active ? '2.5' : '1.8'));
    path.setAttribute('fill', 'none');
    path.style.pointerEvents = 'none';
    svg.appendChild(path);

    if (selected) {
      const delBtn = document.createElementNS('http://www.w3.org/2000/svg', 'g');
      delBtn.style.cursor = 'pointer';
      delBtn.style.pointerEvents = 'auto';
      delBtn.addEventListener('click', (e) => { e.stopPropagation(); deleteEdge(edge.id); });
      const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
      circle.setAttribute('cx', mid);
      circle.setAttribute('cy', midY);
      circle.setAttribute('r', '10');
      circle.setAttribute('fill', '#f7768e');
      const x1 = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      x1.setAttribute('x1', mid - 3.5); x1.setAttribute('y1', midY - 3.5);
      x1.setAttribute('x2', mid + 3.5); x1.setAttribute('y2', midY + 3.5);
      x1.setAttribute('stroke', '#fff'); x1.setAttribute('stroke-width', '2');
      const x2 = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      x2.setAttribute('x1', mid + 3.5); x2.setAttribute('y1', midY - 3.5);
      x2.setAttribute('x2', mid - 3.5); x2.setAttribute('y2', midY + 3.5);
      x2.setAttribute('stroke', '#fff'); x2.setAttribute('stroke-width', '2');
      delBtn.appendChild(circle);
      delBtn.appendChild(x1);
      delBtn.appendChild(x2);
      svg.appendChild(delBtn);
    }
  });
  if (wireDraft) {
    const p1 = portCenter(wireDraft.fromNode, 'out', wireDraft.fromOutput);
    const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
    path.setAttribute('d', 'M ' + p1.x + ' ' + p1.y + ' L ' + wireDraft.x2 + ' ' + wireDraft.y2);
    path.setAttribute('stroke', 'var(--accent)');
    path.setAttribute('stroke-width', '2');
    path.setAttribute('stroke-dasharray', '4 3');
    path.setAttribute('fill', 'none');
    svg.appendChild(path);
  }
}

function renderConfig() {
  const body = document.getElementById('config-body');
  if (!selectedNodeId) {
    body.innerHTML = '<p class="empty-hint">' + nodes.length + ' nodo(s), ' + edges.length + ' conexión(es). Haz clic en uno para ver su resumen.</p>';
    return;
  }
  const node = nodes.find(n => n.id === selectedNodeId);
  if (!node) { body.innerHTML = ''; return; }
  const def = defOf(node.type) || { name: node.type, defaults: {} };
  const fields = Object.entries(def.defaults || {})
    .map(([k, cfg]) => '<div style="margin-bottom:4px;"><b>' + cfg.label + ':</b> ' + (node.config[k] ?? '') + '</div>')
    .join('');
  body.innerHTML =
    '<div style="font-weight:600;margin-bottom:2px;font-size:14px;">' + (node.name || def.name) + '</div>' +
    '<div style="margin-bottom:12px;color:var(--muted);font-size:11px;text-transform:uppercase;font-family:monospace;">' + def.type + '</div>' +
    fields +
    '<button style="margin-top:16px;width:100%;color:var(--err);border-color:rgba(247,118,142,0.3);" onclick="deleteNode(\'' + node.id + '\')">Eliminar nodo</button>' +
    '<p class="palette-hint" style="margin-top:12px">Doble clic sobre el nodo en el canvas para editarlo.</p>';
}

function validateNodeId(value, nodeId) {
  if (!value) return 'El ID no puede estar vacío';
  if (!/^[a-zA-Z][a-zA-Z0-9_-]*$/.test(value)) return 'Solo letras, números, guiones y guiones bajos (debe empezar con letra)';
  if (nodes.some(n => n.id === value && n.id !== nodeId)) return 'Ya existe un nodo con este ID';
  return '';
}

// El HTML del modal llama a estas dos desde oninput (el input de nombre y el
// textarea de notas), pero en esta copia del editor nunca se definieron: el
// rediseño visual las perdió, así que escribir el nombre o las notas de un nodo
// lanzaba ReferenceError y no guardaba nada. Las dos copias desplegadas
// (data-plane y tenant-portal) sí las tienen — esta es la implementación de
// ahí, sin cambios.
function modalUpdateName(value) {
  const node = nodes.find(n => n.id === modalNodeId);
  if (!node) return;
  node.name = value;
  const def = defOf(node.type);
  document.getElementById('modal-title').textContent = value || (def ? def.name : node.type);
  renderCanvas();
}

function modalUpdateNotes(value) {
  const node = nodes.find(n => n.id === modalNodeId);
  if (!node) return;
  node.notes = value;
}

function modalUpdateId(value) {
  const node = nodes.find(n => n.id === modalNodeId);
  if (!node) return;
  const error = validateNodeId(value, node.id);
  const input = document.getElementById('node-id-input');
  const message = document.getElementById('node-id-error');
  if (input) input.classList.toggle('field-error', Boolean(error));
  if (message) message.textContent = error;
  if (error) return;
  const oldId = node.id;
  node.id = value;
  edges.forEach(edge => {
    if (edge.from === oldId) edge.from = value;
    if (edge.to === oldId) edge.to = value;
  });
  if (selectedNodeId === oldId) selectedNodeId = value;
  if (modalNodeId === oldId) modalNodeId = value;
  renderCanvas();
  renderConfig();
}

function renderNodeIdField(node) {
  const error = validateNodeId(node.id, node.id);
  return '<div class="floating-group">' +
    '<input id="node-id-input" class="floating-input ' + (error ? 'field-error' : '') + '" type="text" value="' + node.id + '" placeholder=" " oninput="modalUpdateId(this.value)">' +
    '<label for="node-id-input" class="floating-label">ID Único del Nodo</label>' +
    '<div id="node-id-error" class="field-error-message">' + error + '</div></div>';
}

function renderFieldHtml(key, cfg, node) {
  const val = node.config[key] ?? '';
  const showIfAttr = cfg.showIf ? ' data-showif="' + JSON.stringify(cfg.showIf).replace(/"/g, '&quot;') + '"' : '';

  if (cfg.type === 'keyvalue') {
    let obj = {};
    try { obj = val ? JSON.parse(val) : {}; } catch {}
    modalKeyValueState[key] = Object.entries(obj).map(([k, v]) => ({ key: k, value: v }));
    return '<div' + showIfAttr + ' style="margin-bottom:16px;"><label style="font-size:11px;color:var(--muted);font-weight:600;">' + cfg.label + '</label><div id="kv-' + key + '" class="kv-editor"></div>' +
      '<button type="button" class="kv-add-btn" onclick="addKvRow(\'' + key + '\')">+ Agregar Par</button></div>';
  }
  if (cfg.type === 'rules') {
    let arr = [];
    try { arr = val ? JSON.parse(val) : []; } catch {}
    modalRulesState[key] = arr.length ? arr : [{ property: 'payload', operator: '==', value: '' }];
    return '<div' + showIfAttr + ' style="margin-bottom:16px;"><label style="font-size:11px;color:var(--muted);font-weight:600;">' + cfg.label + '</label><div id="rules-' + key + '" class="rules-editor"></div>' +
      '<button type="button" class="kv-add-btn" onclick="addRuleRow(\'' + key + '\')">+ Agregar Regla</button></div>';
  }
  if (cfg.type === 'configRef') {
    return '<div' + showIfAttr + ' class="floating-group">' +
      '<select id="configref-' + key + '" class="floating-select ' + (val ? 'has-value' : '') + '" onchange="this.classList.toggle(\'has-value\', Boolean(this.value)); modalUpdate(\'' + key + '\', this.value)"><option value="">Cargando…</option></select>' +
      '<label class="floating-label">' + cfg.label + '</label>' +
      '<button type="button" class="kv-add-btn" onclick="openConnectionsPanel(\'' + (cfg.configType || '') + '\')">+ Nueva Conexión</button></div>';
  }
  if (cfg.type === 'textarea' && cfg.code) {
    const versionPickerHtml = key === 'columnsSpec'
      ? '<div class="flexdb-version-row" style="margin-bottom:8px; display:flex; gap:8px;">' +
          '<select id="flexdb-version-select" class="floating-select" style="flex:1;" onchange="applyFlexdbVersionToColumnsSpec(this.value)"><option value="">Versión actual (sin cargar historial)</option></select>' +
          '<button type="button" class="kv-add-btn" style="margin:0;" onclick="refreshFlexdbVersionSelect()">↻ Cargar versiones</button>' +
        '</div>'
      : '';
    return '<div' + showIfAttr + ' style="margin-bottom:16px;"><label style="font-size:11px;color:var(--muted);font-weight:600;">' + cfg.label + '</label>' + versionPickerHtml + '<div class="monaco-box" id="monaco-' + key + '"></div></div>';
  }
  if (cfg.type === 'textarea') {
    return '<div' + showIfAttr + ' class="floating-group">' +
      '<textarea id="field-' + key + '" class="floating-textarea" placeholder=" " oninput="modalUpdate(\'' + key + '\', this.value)">' + val + '</textarea>' +
      '<label for="field-' + key + '" class="floating-label">' + cfg.label + '</label></div>';
  }
  if (cfg.type === 'select') {
    const opts = (cfg.options || []).map(o => '<option value="' + o + '"' + (val === o ? ' selected' : '') + '>' + o + '</option>').join('');
    return '<div' + showIfAttr + ' class="floating-group">' +
      '<select id="field-' + key + '" class="floating-select has-value" onchange="modalUpdate(\'' + key + '\', this.value); applyFieldVisibility();">' + opts + '</select>' +
      '<label for="field-' + key + '" class="floating-label">' + cfg.label + '</label></div>';
  }
  const inputType = cfg.type === 'number' ? 'number' : 'text';
  return '<div' + showIfAttr + ' class="floating-group">' +
    '<input id="field-' + key + '" class="floating-input" type="' + inputType + '" value="' + val + '" placeholder=" " oninput="modalUpdate(\'' + key + '\', this.value)">' +
    '<label for="field-' + key + '" class="floating-label">' + cfg.label + '</label></div>';
}

function openNodeEditor(nodeId) {
  const node = nodes.find(n => n.id === nodeId);
  const def = defOf(node.type);
  if (!node || !def) return;
  modalNodeId = nodeId;
  selectedNodeId = nodeId;

  document.getElementById('modal-title').textContent = node.name || def.name;
  document.getElementById('modal-subtitle').textContent = 'tipo: ' + node.type;
  document.getElementById('modal-routing').textContent = buildRoutingInfo(node);

  const nameFieldHtml =
    '<div class="floating-group">' +
    '<input id="node-name-input" class="floating-input" type="text" value="' + (node.name || '') + '" placeholder=" " oninput="modalUpdateName(this.value)">' +
    '<label for="node-name-input" class="floating-label">Nombre del Nodo (Opcional)</label></div>';

  modalKeyValueState = {};
  modalRulesState = {};
  const fieldsEl = document.getElementById('modal-fields');

  let fieldsBodyHtml;
  if (node.type === 'function') {
    fieldsBodyHtml = renderFunctionFields();
  } else {
    const entries = Object.entries(def.defaults || {});
    const hasTabs = entries.some(([, cfg]) => cfg.tab);
    if (hasTabs) {
      const order = [];
      const groups = {};
      entries.forEach(([key, cfg]) => {
        const tabName = cfg.tab || 'General';
        if (!groups[tabName]) { groups[tabName] = []; order.push(tabName); }
        groups[tabName].push([key, cfg]);
      });
      fieldsBodyHtml =
        '<div class="node-tabs">' +
        order.map((t, i) => '<button type="button" class="node-tab' + (i === 0 ? ' active' : '') + '" data-nodetab="' + t + '" onclick="switchNodeFieldTab(\'' + t + '\')">' + t + '</button>').join('') +
        '</div>' +
        order.map((t, i) => '<div class="node-tab-panel' + (i === 0 ? ' active' : '') + '" id="nodetab-panel-' + t + '">' +
          groups[t].map(([key, cfg]) => renderFieldHtml(key, cfg, node)).join('') +
          '</div>').join('');
    } else {
      fieldsBodyHtml = entries.map(([key, cfg]) => renderFieldHtml(key, cfg, node)).join('');
    }
  }
  fieldsEl.innerHTML = renderNodeIdField(node) + nameFieldHtml + fieldsBodyHtml;

  applyFieldVisibility();

  Object.entries(def.defaults || {}).forEach(([key, cfg]) => {
    if (cfg.type === 'keyvalue') renderKvRows(key);
    if (cfg.type === 'rules') renderRulesRows(key);
    if (cfg.type === 'configRef') mountConfigRefField(key, cfg, node);
  });

  disposeMonacoEditors();
  if (node.type === 'function') {
    mountFnEditor('onStart', def, node);
  } else {
    const codeFields = Object.entries(def.defaults || {}).filter(([, cfg]) => cfg.code);
    if (codeFields.length) {
      loadMonaco().then((monaco) => {
        if (modalNodeId !== nodeId) return;
        codeFields.forEach(([key, cfg]) => {
          const container = document.getElementById('monaco-' + key);
          if (!container) return;
          const editor = monaco.editor.create(container, {
            value: node.config[key] ?? cfg.value ?? '',
            language: 'javascript',
            theme: 'vs-dark',
            minimap: { enabled: false },
            automaticLayout: true,
            fontSize: 13,
            scrollBeyondLastLine: false,
          });
          editor.onDidChangeModelContent(() => modalUpdate(key, editor.getValue()));
          activeMonacoEditors[key] = editor;
        });
      });
    }
  }

  const notesEl = document.getElementById('modal-notes');
  if (notesEl) notesEl.value = node.notes || '';
  switchModalTab('props');

  const t = document.getElementById('modal-test-result');
  if (t) {
    t.classList.remove('show', 'error');
    t.textContent = '';
  }

  document.getElementById('modal-overlay').classList.add('open');
  renderCanvas();
}

function buildRoutingInfo(node) {
  const incoming = edges.filter(e => e.to === node.id).length;
  const outgoing = edges.filter(e => e.from === node.id).length;
  let txt = '';
  if (incoming) txt += incoming + ' conexión entrante. ';
  if (outgoing) txt += outgoing + ' conexión saliente. ';
  if (!txt) txt = 'Sin conexiones. ';
  txt += 'Esta acción se ejecuta dentro del flujo "' + flowName + '".';
  return txt;
}

function switchFnTab(tab) {
  document.querySelectorAll('.fn-tab').forEach(el => el.classList.toggle('active', el.dataset.fn === tab));

function mountFnEditor(key, def, node) {
  if (activeMonacoEditors[key]) return;
  const cfg = def.defaults[key];
  if (!cfg) return;
  loadMonaco().then((monaco) => {
    if (modalNodeId !== node.id) return;
    if (activeMonacoEditors[key]) return;
    const container = document.getElementById('monaco-' + key);
    if (!container) return;
    const editor = monaco.editor.create(container, {
      value: node.config[key] ?? cfg.value ?? '',
      language: 'javascript',
      theme: 'vs-dark',
      minimap: { enabled: false },
      automaticLayout: true,
      fontSize: 13,
      scrollBeyondLastLine: false,
    });
    editor.onDidChangeModelContent(() => modalUpdate(key, editor.getValue()));
    activeMonacoEditors[key] = editor;
  });
}

let monacoLoadPromise = null;
let activeMonacoEditors = {};

function loadMonaco() {
  if (monacoLoadPromise) return monacoLoadPromise;
  const BASE = 'https://cdn.jsdelivr.net/npm/monaco-editor@0.45.0/min/vs';
  monacoLoadPromise = new Promise((resolve) => {
    const script = document.createElement('script');
    script.src = BASE + '/loader.js';
    script.onload = () => {
      window.require.config({ paths: { vs: BASE } });
      window.require(['vs/editor/editor.main'], () => {
        monaco.languages.typescript.javascriptDefaults.addExtraLib(
          `declare const msg: { payload: any; topic?: string; [key: string]: any };
           declare const ctx: { env: Record<string, any>; flow: Record<string, any>; log: (...args: any[]) => void };
           declare const context: Record<string, any>;`,
          'flow-engine-globals.d.ts'
        );
        monaco.languages.typescript.javascriptDefaults.setCompilerOptions({
          allowNonTsExtensions: true,
          allowJs: true,
          checkJs: false,
          target: monaco.languages.typescript.ScriptTarget.ES2020,
        });
        resolve(window.monaco);
      });
    };
    document.head.appendChild(script);
  });
  return monacoLoadPromise;
}

function disposeMonacoEditors() {
  Object.values(activeMonacoEditors).forEach((ed) => ed.dispose());
  activeMonacoEditors = {};

function applyFieldVisibility() {

function flexdbClientSlugify(name) {
  return String(name || '')
    .trim()
    .toLowerCase()
    .normalize('NFD').replace(/[̀-ͯ]/g, '')
    .replace(/[^a-z0-9]+/g, '_')
    .replace(/^_+|_+$/g, '');
}

// --- conversión hacia/desde el flow.json canónico ---

function toFlowJson() {
  return nodes.map(n => {
    const def = defOf(n.type) || { outputs: 1 };
    const outputs = outputCountFor(n, def);
    const wires = [];
    for (let i = 0; i < outputs; i++) {
      wires.push(edges.filter(e => e.from === n.id && (e.fromOutput || 0) === i).map(e => e.to));
    }
    return { id: n.id, type: n.type, name: n.name || '', x: n.position.x, y: n.position.y, notes: n.notes || '', ...n.config, wires };
  });
}

function migrateLegacyIds(flowJson) {
  const legacyNodes = flowJson.filter(n => /^n_\d+$/.test(String(n.id)) || /^n\d+$/.test(String(n.id)));
  if (!legacyNodes.length) return flowJson;
  if (!confirm('Este flow usa IDs antiguos (n_1, n_2...). ¿Migrar a UUIDs?')) return flowJson;
  const idMap = new Map(legacyNodes.map(n => [n.id, uid('n')]));
  return flowJson.map(n => ({
    ...n,
    id: idMap.get(n.id) || n.id,
    wires: (n.wires || []).map(targets => targets.map(target => idMap.get(target) || target)),
  }));
}

function loadFlowJsonInto(flowJson) {
  flowJson = migrateLegacyIds(flowJson);
  nodes = [];
  edges = [];
  selectedNodeId = null;

  flowJson.forEach(n => {
    const def = defOf(n.type);
    if (!def) { logLine('❌ tipo de nodo desconocido: ' + n.type, true); return; }
    const preset = { __id: n.id, __notes: n.notes || '', __name: n.name || '' };
    Object.keys(def.defaults || {}).forEach(k => { preset[k] = n[k]; });
    addNode(n.type, n.x || 100, n.y || 100, preset);
  });

  flowJson.forEach(n => {
    (n.wires || []).forEach((targets, outIndex) => {
      targets.forEach(targetId => {
        if (!nodes.find(x => x.id === n.id) || !nodes.find(x => x.id === targetId)) return;
        edges.push({ id: uid('e'), from: n.id, fromOutput: outIndex, to: targetId });
      });
    });
  });

  resyncIdCounterFromLoadedIds([...nodes.map(n => n.id), ...edges.map(e => e.id)]);

  selectedNodeId = null;
  renderCanvas();
  renderConfig();
  markFlowSaved();
}

// --- acciones de la barra superior ---

async function loadFlow(name) {
  if (isFlowDirty()) {
    const r = await Swal.fire({
      icon: 'warning',
      title: 'Cambios sin guardar',
      text: 'Vas a cambiar de flujo y hay cambios que no se han guardado en el servidor. Si continúas, se pierden.',
      showCancelButton: true,
      confirmButtonText: 'Descartar y continuar',
      cancelButtonText: 'Cancelar',
      confirmButtonColor: '#e06c75',
    });
    if (!r.isConfirmed) return;
  }
  flowName = name || flowName || 'example';
  const flowJson = await apiFetch('/_editor/api/flows/' + flowName);
  if (!flowJson || flowJson.error) { logLine('⚠ no existe el flow "' + flowName + '" todavía', true); return; }
  loadFlowJsonInto(flowJson);
  logLine('✔ flujo "' + flowName + '" cargado');
  setStatus('Flujo "' + flowName + '" cargado.', nodes.length + ' nodo(s)');
}

async function saveFlow() {
  const flowJson = toFlowJson();
  await apiFetch('/_editor/api/flows/' + flowName, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(flowJson),
  });
  markFlowSaved();
  logLine('✔ flujo "' + flowName + '" guardado');
  setStatus('Flujo "' + flowName + '" guardado.', '');
}

function clearCanvas() {
  nodes = [];
  edges = [];
  selectedNodeId = null;
  renderCanvas();
  renderConfig();
}

async function runFlow(entryId) {
  const flowJson = toFlowJson();
  if (!flowJson.length) { setStatus('No hay nodos para ejecutar.', ''); return; }
  setStatus('Ejecutando flujo "' + flowName + '"…', '');
  const data = await apiFetch('/_editor/api/run', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ flow: flowJson, entryId: entryId || null, flowName }),
  });
  if (data.error) { logLine('❌ ' + data.error, true); setStatus('Error al ejecutar.', ''); return; }
  (data.logs || []).forEach(l => logLine(l));
  logLine('🐛 [resultado] ' + JSON.stringify(data.result));
  setStatus('Flujo "' + flowName + '" ejecutado.', 'completado');
  switchConfigPanelTab('debug');
}

function triggerEntryNode(nodeId) {
  runFlow(nodeId);
}

document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape') deselectEdge();
  
  const isEditingText = ['INPUT', 'TEXTAREA', 'SELECT'].includes(document.activeElement.tagName) || 
                        document.activeElement.isContentEditable;

  if ((e.key === 'Delete' || e.key === 'Backspace') && !isEditingText) {
    if (selectedEdgeId) {
      deleteEdge(selectedEdgeId);
    } else if (selectedNodeId) {
      deleteNode(selectedNodeId);
    }
  }
});
