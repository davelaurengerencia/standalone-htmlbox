  <script>

    // ============ Core: debug, apiFetch, reporte de errores ============
    // Se carga primero dentro de este mismo <script> — todo lo de abajo
    // (portalApp()/dataTab() o adminApp()) puede usar debugLog()/apiFetch().
    //
    // window.HTMLBOX_ENV y window.HTMLBOX_*_VERSION los inyecta el Worker
    // (ver renderShell() en lib/partials.js) leyendo sus vars de wrangler.jsonc.

    const HBX_DEBUG = (window.HTMLBOX_ENV || 'production') !== 'production'

    function debugLog(...args) {
      if (HBX_DEBUG) console.log('[HTMLBox]', ...args)
    }

    // Manda un error del browser al control-plane para que quede en Workers
    // Logs del lado servidor, no solo en la consola del usuario (que nadie va
    // a ver). Best-effort: si el propio reporte falla, no hace nada más — un
    // error de reporting nunca debe generar un error nuevo.
    function reportClientError(payload) {
      try {
        fetch('/api/client-error', {
          method: 'POST',
          credentials: 'include',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            message: String(payload.message || '').slice(0, 2000),
            stack: String(payload.stack || '').slice(0, 4000),
            url: window.location.href,
            userAgent: navigator.userAgent,
          }),
          keepalive: true,
        }).catch(() => {})
      } catch { /* nunca romper por esto */ }
    }

    window.addEventListener('error', (e) => {
      debugLog('window.onerror:', e.error || e.message)
      reportClientError({ message: e.message, stack: e.error && e.error.stack })
    })

    window.addEventListener('unhandledrejection', (e) => {
      const reason = e.reason
      debugLog('unhandledrejection:', reason)
      reportClientError({
        message: reason && reason.message ? reason.message : String(reason),
        stack: reason && reason.stack,
      })
    })

    // Wrapper centralizado para llamadas a /api/*. Maneja de forma consistente:
    // status no-2xx, respuesta que no es JSON válido, y errores de red — y
    // adjunta el cf-ray de la respuesta (si vino) al error, para poder buscar
    // esa request exacta en los logs de Cloudflare en vez de adivinar.
    //
    // Uso: const data = await apiFetch('/api/boxes', { method: 'POST', body: {...} })
    // El error lanzado trae .status, .code (si el server mandó {error:'code'}) y .rayId.
    async function apiFetch(path, opts = {}) {
      const { body, headers, ...rest } = opts
      const init = {
        credentials: 'include',
        headers: { 'Content-Type': 'application/json', ...(headers || {}) },
        ...rest,
      }
      if (body !== undefined) init.body = typeof body === 'string' ? body : JSON.stringify(body)

      let res
      try {
        res = await fetch(path, init)
      } catch (err) {
        const netErr = new Error('Sin conexión con el servidor.')
        netErr.cause = err
        throw netErr
      }

      const rayId = res.headers.get('cf-ray') || null
      let data = null
      try {
        data = await res.json()
      } catch {
        // Respuesta no era JSON (ej. un 500 en HTML del edge) — seguimos con data=null.
      }

      if (!res.ok) {
        const err = new Error((data && (data.error || data.message)) || `Error ${res.status}`)
        err.status = res.status
        err.code = data && data.error
        err.rayId = rayId
        throw err
      }

      return data
    }

    debugLog('boot', window.HTMLBOX_PORTAL_VERSION || window.HTMLBOX_ADMIN_VERSION || 'dev')

    // Regex del tenant slug — mirror del backend (packages/shared/src/id.js#TENANT_SLUG_REGEX).
    // Si cambia allá, cambiar acá también.
    const TENANT_SLUG_REGEX = /^[a-z][a-z0-9-]{1,30}[a-z0-9]$/

    function adminApp() {
      return {
        user: null,
        email: '',
        busy: false,
        lastMessage: '',
        previewLink: null,

        // Tabs
        activeTab: 'tenants',

        // Stats agregadas del dashboard
        stats: null,
        // Tenants con counts (workspace_count, box_count, wfp breakdown)
        adminTenants: [],

        // Modal nuevo tenant (platform owner)
        newTenantModalOpen: false,
        creatingTenant: false,
        newTenantError: '',
        newTenant: { name: '', slug: '' },
        newTenantSlugTouched: false,
        tenantSlugValid: false,

        // Modal detalle de un tenant (workspaces + boxes)
        tenantDetailModalOpen: false,
        currentTenant: null,
        tenantDetail: null,
        loadingTenantDetail: false,

        // Backward compat — la versión vieja del dashboard usaba
        // tenants/currentTenant/workspaces. Los mantenemos por si hay
        // otros partials o código viejo que los lea. El dashboard nuevo
        // usa adminTenants + tenantDetail.
        tenants: [],
        workspaces: [],

        toast: null,
        _toastTimer: null,

        async init() {
          await this.loadMe()
        },

        async loadMe() {
          try {
            const data = await apiFetch('/api/auth/me')
            this.user = data.user
            if (this.user) {
              await Promise.all([this.loadStats(), this.loadAdminTenants()])
            }
          } catch (err) {
            debugLog('loadMe:', err)
          }
        },

        async loadStats() {
          try {
            this.stats = await apiFetch('/api/admin/stats')
          } catch (err) {
            debugLog('loadStats:', err)
          }
        },

        async loadAdminTenants() {
          try {
            const data = await apiFetch('/api/admin/tenants')
            this.adminTenants = data.tenants ?? []
            // Mantener el shape viejo también — usado por código que
            // ya estuviera referenciando tenants[].
            this.tenants = this.adminTenants
          } catch (err) {
            debugLog('loadAdminTenants:', err)
          }
        },

        async requestMagicLink() {
          this.busy = true
          this.lastMessage = ''
          this.previewLink = null
          try {
            const data = await apiFetch('/api/auth/request', {
              method: 'POST',
              body: { email: this.email },
            })
            this.lastMessage = data.message || 'Listo. Revisá tu email.'
            if (data._dev_preview) this.previewLink = data._dev_preview
          } catch (err) {
            this.lastMessage = 'Error: ' + err.message
          } finally {
            this.busy = false
          }
        },

        async logout() {
          try {
            await apiFetch('/api/auth/logout', { method: 'POST' })
          } catch (err) {
            debugLog('logout:', err)
          }
          this.user = null
          this.tenants = []
          this.adminTenants = []
          this.currentTenant = null
          this.tenantDetail = null
          this.workspaces = []
          this.stats = null
        },

        // --------- Modal nuevo tenant (platform owner) ---------

        openNewTenantModal() {
          if (!this.user?.is_platform_owner) {
            this.showToast('Solo el platform owner puede crear tenants.')
            return
          }
          this.newTenant = { name: '', slug: '' }
          this.newTenantError = ''
          this.newTenantSlugTouched = false
          this.tenantSlugValid = false
          this.newTenantModalOpen = true
        },
        closeNewTenantModal() {
          this.newTenantModalOpen = false
          this.newTenantError = ''
          this.creatingTenant = false
        },

        onTenantNameInput(event) {
          this.newTenant.name = event.target.value
          if (!this.newTenantSlugTouched) {
            this.newTenant.slug = this._tenantSlugifyFromName(this.newTenant.name)
            this._recomputeTenantSlugValid()
          }
        },
        onTenantSlugInput(event) {
          this.newTenant.slug = event.target.value
          this.newTenantSlugTouched = true
          this._recomputeTenantSlugValid()
        },
        _recomputeTenantSlugValid() {
          this.tenantSlugValid = TENANT_SLUG_REGEX.test(this.newTenant.slug)
        },
        _tenantSlugifyFromName(name) {
          return String(name || '')
            .toLowerCase()
            .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
            .replace(/[^a-z0-9_-]+/g, '-')
            .replace(/^-+|-+$/g, '')
            .slice(0, 60) || ''
        },

        async createTenant() {
          if (!this.user?.is_platform_owner) {
            this.newTenantError = 'Solo el platform owner puede crear tenants.'
            return
          }
          const name = this.newTenant.name.trim()
          const slug = this.newTenant.slug.trim()
          if (!name) {
            this.newTenantError = 'Poné un nombre.'
            return
          }
          if (!this.tenantSlugValid) {
            this.newTenantError = 'Slug inválido.'
            return
          }
          this.creatingTenant = true
          this.newTenantError = ''
          try {
            const data = await apiFetch('/api/tenants', {
              method: 'POST',
              body: { name, slug },
            })
            if (!data || data.error) {
              const code = data?.error
              this.newTenantError = ({
                invalid_slug: 'Slug inválido.',
                slug_taken: 'Ya hay un tenant con ese slug.',
                missing_name: 'Falta el nombre.',
                platform_owner_only: 'Tu cuenta no es platform owner.',
              })[code] || ('Error: ' + (code || 'desconocido'))
              return
            }
            this.showToast(`Tenant "${data.tenant.name}" creado.`)
            this.closeNewTenantModal()
            await Promise.all([this.loadStats(), this.loadAdminTenants()])
          } catch (err) {
            this.newTenantError = 'Error: ' + (err?.code || err?.message || 'desconocido')
          } finally {
            this.creatingTenant = false
          }
        },

        // --------- Modal detalle de un tenant ---------

        async openTenantDetail(t) {
          this.currentTenant = t
          this.tenantDetailModalOpen = true
          this.tenantDetail = null
          this.loadingTenantDetail = true
          try {
            // Traer workspaces + boxes en paralelo. (boxes no tiene endpoint
            // cross-tenant en /api/boxes/:workspaceId/...; usamos /api/boxes
            // con ?workspace=... no aplica porque tenant tiene varios.
            // Para v1: leemos workspaces del tenant + boxes via /api/boxes
            // una vez por workspace.)
            const wsData = await apiFetch(`/api/tenants/${t.id}/workspaces`)
            const workspaces = wsData.workspaces ?? []
            const boxesArrays = await Promise.all(workspaces.map((w) =>
              apiFetch(`/api/boxes?workspace=${w.id}`).then((d) => d.boxes ?? [])
                .catch(() => [])
            ))
            this.tenantDetail = {
              workspaces,
              boxes: boxesArrays.flat(),
            }
          } catch (err) {
            debugLog('openTenantDetail:', err)
            this.showToast('Error cargando detalle: ' + (err?.code || err?.message))
          } finally {
            this.loadingTenantDetail = false
          }
        },
        closeTenantDetail() {
          this.tenantDetailModalOpen = false
          // Mantener currentTenant/tenantDetail un momento para que la
          // animación de cierre tenga datos, pero no es estrictamente
          // necesario — Alpine limpia solo.
        },

        // --------- Utilidades ---------

        // '2026-08-27T...' → 'hace 5 min' / 'hace 2h' / 'hace 3d' / fecha completa
        formatRelativeTime(iso) {
          if (!iso) return '—'
          const t = new Date(iso + (iso.endsWith('Z') ? '' : 'Z'))
          if (isNaN(t.getTime())) return iso
          const diffMs = Date.now() - t.getTime()
          const sec = Math.round(diffMs / 1000)
          if (sec < 60) return 'hace ' + sec + 's'
          const min = Math.round(sec / 60)
          if (min < 60) return 'hace ' + min + ' min'
          const hr = Math.round(min / 60)
          if (hr < 24) return 'hace ' + hr + 'h'
          const day = Math.round(hr / 24)
          if (day < 30) return 'hace ' + day + 'd'
          // Más viejo: fecha completa en formato YYYY-MM-DD.
          return t.toISOString().slice(0, 10)
        },

        showToast(msg) {
          this.toast = msg
          clearTimeout(this._toastTimer)
          this._toastTimer = setTimeout(() => { this.toast = null }, 3000)
        },
      }
    }
  </script>
