    // Manda un error del browser al control-plane para que quede en Workers
    // Logs del lado servidor, no solo en la consola del usuario (que nadie va
    // a ver). Best-effort: si el propio reporte falla, no hace nada más — un
    // error de reporting nunca debe generar un error nuevo.
    function reportClientError(payload) {
      try {
        fetch('/api/client-error', {
          method: 'POST',
          credentials: 'include',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            message: String(payload.message || '').slice(0, 2000),
            stack: String(payload.stack || '').slice(0, 4000),
            url: window.location.href,
            userAgent: navigator.userAgent,
          }),
          keepalive: true,
        }).catch(() => {})
      } catch { /* nunca romper por esto */ }
    }

    window.addEventListener('error', (e) => {
      debugLog('window.onerror:', e.error || e.message)
      reportClientError({ message: e.message, stack: e.error && e.error.stack })
    })

    window.addEventListener('unhandledrejection', (e) => {
      const reason = e.reason
      debugLog('unhandledrejection:', reason)
      reportClientError({
        message: reason && reason.message ? reason.message : String(reason),
        stack: reason && reason.stack,
      })
    })

    // Wrapper centralizado para llamadas a /api/*. Maneja de forma consistente:
    // status no-2xx, respuesta que no es JSON válido, y errores de red — y
    // adjunta el cf-ray de la respuesta (si vino) al error, para poder buscar
    // esa request exacta en los logs de Cloudflare en vez de adivinar.
    //
    // Uso: const data = await apiFetch('/api/boxes', { method: 'POST', body: {...} })
    // El error lanzado trae .status, .code (si el server mandó {error:'code'}) y .rayId.
    async function apiFetch(path, opts = {}) {
      const { body, headers, ...rest } = opts
      const init = {
        credentials: 'include',
        headers: { 'Content-Type': 'application/json', ...(headers || {}) },
        ...rest,
      }
      if (body !== undefined) init.body = typeof body === 'string' ? body : JSON.stringify(body)

      let res
      try {
        res = await fetch(path, init)
      } catch (err) {
        const netErr = new Error('Sin conexión con el servidor.')
        netErr.cause = err
        throw netErr
      }

      const rayId = res.headers.get('cf-ray') || null
      let data = null
      try {
        data = await res.json()
      } catch {
        // Respuesta no era JSON (ej. un 500 en HTML del edge) — seguimos con data=null.
      }

      if (!res.ok) {
        const err = new Error((data && (data.error || data.message)) || `Error ${res.status}`)
        err.status = res.status
        err.code = data && data.error
        err.rayId = rayId
        throw err
      }

      return data
    }

    // ============ Editor (htmlbox-spec-codemirror-editor.md) ============
    //
    // Reemplaza el anterior split-view (Todo | JS | CSS). CodeMirror 6
    // con @codemirror/lang-html entiende HTML+<script>+<style> embebido
    // nativamente — un único editor, sin vistas parciales que reensamblar.
    // Helpers puros que viven fuera de portalApp() para ser testeables en
    // Node vía vm sandbox; CM6 en sí se carga vía dynamic import (esm.sh)
    // cuando el usuario abre la tab editor.
    //
    // Por qué no usar x-model con CM: CM mantiene su propio state
    // (undo stack, cursor, syntax). Forzarlo a re-renderizar en cada
    // cambio reactivo resetea el cursor/undo. Lo correcto es empujar sus
    // cambios a editorHtml vía updateListener y sincronizar de afuera solo
    // cuando HTML cambia por fuera (cargar box, IA, formatear).

    // Re-indent simple del HTML — fallback defensivo si js-beautify no
    // carga. NO es un beautifier completo (no entiende HTML+JS+CSS embebido
    // tan bien como js-beautify). Testeado en __tests__/codemirrorEditor.test.js.
    function reindentHtml(html) {
      let level = 0
      const out = []
      const tokens = (html || '')
        .replace(/></g, '>\n<')
        .replace(/>\s+</g, '>\n<')
        .split('\n')
      for (const t of tokens) {
        const trimmed = t.trim()
        if (!trimmed) continue
        if (/^<\/.+>/.test(trimmed)) level = Math.max(0, level - 1)
        out.push('  '.repeat(level) + trimmed)
        if (/^<[^!?\/][^>]*[^/]>$/.test(trimmed) && !/^<(br|hr|img|input|meta|link)/.test(trimmed)) {
          level++
        }
      }
      return out.join('\n')
    }

            ta.select()
            document.execCommand('copy')

            this.$refs.teamTabRef.loadTeam()
          }

        async handleFileUpload(ev) {
          const file = ev.target.files[0]
          if (!file) return
          const reader = new FileReader()
          reader.onload = async (e) => {
            const content = e.target.result
            if (!this.currentBox) {
              this.showToast('Seleccioná o creá un box antes de subir.')
              ev.target.value = ''
              return

            this._resetEditorState(content)
            this.activeTab = 'editor'
            this.showToast(`HTML cargado (${content.length} bytes). Tocá Guardar para crear v${(this.currentBox?.htmlbox_version ?? 0) + 1}.`)
          }
          reader.readAsText(file)
          ev.target.value = ''
        },

            this.showToast(`HTML cargado (${this.newBox.htmlContent.length} bytes).`)
          }
          reader.readAsText(file)
          ev.target.value = ''

        async saveCurrentBox() {
          if (!this.currentBox) return
          // editorHtml SIEMPRE es el documento completo (un único editor de
          // HTML con resaltado de <script>/<style> embebidos; ver
          // htmlbox-spec-codemirror-editor.md). Sin vista parcial.
          const fullHtml = this.editorHtml || ''
          if (!fullHtml.trim()) {
            this.showToast('El HTML está vacío.')
            return
          }
          this.saving = true
          try {
            const size = new Blob([fullHtml]).size
            const urlData = await apiFetch(`/api/boxes/${this.currentBox.id}/upload-url`, {
              method: 'POST',
              body: { size, contentType: 'text/html' },
            })

            const putRes = await fetch(urlData.uploadUrl, {
              method: 'PUT',
              headers: { 'Content-Type': 'text/html' },
              body: fullHtml,
            })
            if (!putRes.ok) throw new Error(`PUT R2 falló: ${putRes.status}`)

            const summary = prompt('Resumen corto de la versión (obligatorio):', 'Cambios manuales')
            if (!summary) {
              this.showToast('Versión cancelada.')
              return
            }
            const confData = await apiFetch(`/api/boxes/${this.currentBox.id}/html`, {
              method: 'POST',
              body: { r2Key: urlData.key, summary, source: 'portal' },
            })

            await this.loadBoxes()
            const fresh = this.boxes.find((b) => b.id === this.currentBox.id)
            if (fresh) await this.selectBox(fresh)
            this.activeTab = 'preview'
            this.showToast(`Guardado como v${confData.version}.`)
            if (this.currentBox && this.currentBox.auto_analyze_on_save) {
              this.aiAutoAnalyzePending = true
              this.aiAnalyze({ silent: true }).then(() => {
                this.aiAutoAnalyzePending = false
              })
            }
          } catch (err) {
            debugLog('saveCurrentBox:', err)
            this.showToast('Error: ' + (err.code || err.message))
          } finally {
            this.saving = false
          }
        },

        onTenantNameInput(event) {

          if (!this.newTenantSlugTouched) {

        _recomputeTenantSlugValid() {
          this.tenantSlugValid = /^[a-z][a-z0-9-]{1,30}[a-z0-9]$/.test(this.newTenant.slug)
        },

        // Duplicado lógico de packages/shared/src/slugify.js — duplicamos
        // intencionalmente para que el bundle del portal pueda servir
        // sin importar @htmlbox/shared (que hoy no se importa desde el
        // portal; si en el futuro el portal empieza a usar el shared,
        // podemos borrar esta y reusar). Mismas reglas.
        _tenantSlugifyFromName(name) {
          return String(name || '')
            .toLowerCase()
            .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
            .replace(/[^a-z0-9_-]+/g, '-')
            .replace(/^-+|-+$/g, '')
            .slice(0, 60) || ''
        },

        async createTenant() {
          if (!this.user?.is_platform_owner) {
            this.newTenantError = 'Solo el platform owner puede crear tenants.'
            return
          }
          const name = this.newTenant.name.trim()
          const slug = this.newTenant.slug.trim()
          if (!name) {
            this.newTenantError = 'Poné un nombre.'
            return
          }
          if (!this.tenantSlugValid) {
            this.newTenantError = 'Slug inválido. Debe ser minúsculas, números y guiones, empezar con letra (3-32 chars).'
            return
          }
          this.creatingTenant = true
          this.newTenantError = ''
          try {
            const data = await apiFetch('/api/tenants', {
              method: 'POST',
              body: { name, slug },
            })
            if (!data || data.error) {
              const code = data?.error
              this.newTenantError = ({
                invalid_slug: 'Slug inválido. Mirá las reglas abajo del input.',
                slug_taken: 'Ya hay un tenant con ese slug. Probá otro.',
                missing_name: 'Falta el nombre.',
                platform_owner_only: 'Tu cuenta no es platform owner.',

        formatBytes(n) {

// Reset unificado cuando se asigna un HTML nuevo al editor (cargar
        // box, subir archivo, crear box, logout, vaciar lista). Si el
        // editor ya montó, sincroniza su buffer; si no, no hace nada (el

// --- Editor CodeMirror 6 (htmlbox-spec-codemirror-editor.md) ---
        //
        // mountCMIfNeeded() corre cada vez que activeTab cambia (vía
        // x-effect en el <div>) y monta CM la primera vez que la tab
        // editor está visible. NO usar x-init: la tab arranca oculta con
        // display:none y montar sobre un contenedor sin dimensiones es
        // frágil. Idempotente — x-effect re-corre en cada cambio pero
        // el guard corta.
        mountCMIfNeeded() {
          if (this._cmMounted) return
          if (this.activeTab !== 'editor') return
          const container = this.$refs && this.$refs.cmContainer
          if (!container) return
          this._cmMounted = true
          this.initCM(container)
        },

        // initCM() hace el mount puro. CM6 se distribuye como subpackages
        // individuales (@codemirror/state, @codemirror/view, etc.); el
        // meta-package `codemirror` solo tiene 6.0.x y 6.65.7 en npm, no
        // llega a 6.65.1 — por eso importamos cada parte desde su
        // subpackage. Importamos 5 módulos en paralelo: state, view,
        // commands (keymaps), lang-html, theme one-dark. Sin usar arrow
        // functions en el updateListener — Alpine proxy-ea 'this'.
        async initCM(container) {
          this._cmReady = (async () => {
            const [
              { EditorState },
              { EditorView, lineNumbers, keymap },
              { defaultKeymap, history, historyKeymap },
              { html },
              { oneDark },
            ] = await Promise.all([
              import('https://esm.sh/@codemirror/state@6.7.1'),
              import('https://esm.sh/@codemirror/view@6.43.9'),
              import('https://esm.sh/@codemirror/commands@6.11.0'),
              import('https://esm.sh/@codemirror/lang-html@6.4.12'),
              import('https://esm.sh/@codemirror/theme-one-dark@6.1.3'),
            ])

            // CM → Alpine: solo en docChanged (no en selección / scroll /
            // otras transacciones) para no spamear reactivity. Bind
            // explícito porque el callback corre fuera del proxy de Alpine.
            const updateListener = EditorView.updateListener.of(function (update) {
              if (update.docChanged) {
                this.editorHtml = update.state.doc.toString()
              }
            }.bind(this))

            const state = EditorState.create({
              doc: this.editorHtml || '',
              extensions: [
                lineNumbers(),
                history(),
                keymap.of([defaultKeymap, historyKeymap]),
                // lang-html() reconoce <script>/<style> embebidos y los
                // tokeniza con JS/CSS automáticamente — no hay que
                // registrar lenguajes aparte.
                html(),
                oneDark,
                EditorView.theme({
                  // Layout flexbox-style para que el scroller ocupe
                  // exactamente el alto del wrapper. Sin esto, CM mide
                  // el alto por su contenido y overflow nunca se
                  // dispara. Crítico el display:flex + flexDirection
                  // column — el wrapper es hijo de un flex item y
                  // necesita propagar altura disponible al scroller
                  // (no al contenido directo).
                  '&': {
                    height: '100%',
                    display: 'flex',
                    flexDirection: 'column',
                    fontSize: '13px',
                  },
                  '.cm-scroller': {
                    flex: '1 1 0',
                    minHeight: '0',
                    overflow: 'auto',
                  },
                  '.cm-content': { minHeight: '0' },
                  '.cm-gutters': { backgroundColor: 'transparent', borderRight: '1px solid rgba(255,255,255,0.05)' },
                }, { dark: true }),
                updateListener,
              ],
            })

            this._cmEditorView = new EditorView({ state, parent: container })
          })()
          await this._cmReady
        },

        // Alpine → CM: llamarlo cuando HTML cambia por fuera del editor
        // (carga de box, IA, formatear). No-op seguro si CM todavía no
        // montó (primera carga antes de que _cmReady resuelva).
        _syncCMValue(html) {
          if (!this._cmEditorView) return
          const view = this._cmEditorView
          const current = view.state.doc.toString()
          const next = html || ''
          if (current === next) return
          view.dispatch({
            changes: { from: 0, to: current.length, insert: next },
          })
        },

        get previewSrcdoc() {
          const html = this.editorHtml || ''
          if (!html) return ''
          const sdkTag = '<script src="/_sdk/htmlbox.js?boxId=preview&v=public"><\/script>'
          if (/<\/body\s*>/i.test(html)) {
            return html.replace(/<\/body\s*>/i, `${sdkTag}</body>`)
          }
          return html + '\n' + sdkTag
        },

        formatHTML() {
          // js-beautify (UMD cargado en shell.html.txt) maneja HTML+JS+CSS
          // embebido decentemente. Si el CDN está caído y window.html_beautify
          // no existe, caemos a reindentHtml offline (best-effort).
          if (typeof window.html_beautify === 'function') {
            try {
              this.editorHtml = window.html_beautify(this.editorHtml || '', {
                indent_size: 2,
                wrap_line_length: 0,
                preserve_newlines: true,
              })
              this._syncCMValue(this.editorHtml)
              this.showToast('HTML formateado.')
              return
            } catch (err) {
              console.warn('js-beautify falló, usando reindentHtml:', err)
            }
          }
          // Fallback offline.
          this.editorHtml = reindentHtml(this.editorHtml || '')
          this._syncCMValue(this.editorHtml)
          this.showToast('HTML re-indentado (fallback — js-beautify no cargó).')
        },

        showToast(msg, duration = 3000) {
          this.toast = msg
          clearTimeout(this.toastTimer)
          this.toastTimer = setTimeout(() => { this.toast = null }, duration)
        },

        // --------- AI (fase 4 v1) ---------

        async aiAnalyze({ silent = false } = {}) {
          if (!this.currentBox) {
            if (!silent) this.showToast('Seleccioná un box antes de analizar.')
            return
          }
          // La IA necesita el HTML completo. editorHtml directamente — no hay
          // vista parcial.
          const fullHtml = this.editorHtml || ''
          if (!fullHtml || !fullHtml.trim()) {
            if (!silent) this.showToast('Escribí o pegá HTML antes de analizar.')
            return
          }
          this.aiAnalyzing = true
          if (silent) this.aiAutoAnalyzePending = true
          try {
            const data = await apiFetch(`${CONTROL_PLANE}/api/ai/analyze-html`, {
              method: 'POST',
              body: { boxId: this.currentBox.id, html: fullHtml },
            })
            if (!data.tables || data.tables.length === 0) {
              this.showToast('La IA no propuso tablas para este HTML.')
              return
            }
            this.aiSchema = data
            // Decoramos cada tabla con _candidateRowCount — la cantidad de
            // filas REALES que detectamos en el HTML (no las sample_rows de la
            // IA). El modal usa esto para mostrar el badge "Aplicar migra
            // datos — N filas" vs "Solo propuesta de schema".
            const candidatesByName = new Map((data.candidates || []).map((c) => [c.varName, c]))
            const edited = JSON.parse(JSON.stringify(data.tables))
            for (const t of edited) {
              if (t.source_var && candidatesByName.has(t.source_var)) {
                t._candidateRowCount = candidatesByName.get(t.source_var).rowCount
              } else {
                t.source_var = null  // defensivo: si el server no mandó source_var, lo limpiamos
                t._candidateRowCount = 0
              }
            }
            this.aiModalEdited = edited
            if (silent) {
              this.aiPendingReview = true
              this.showToast(`IA propuso ${data.tables.length} tabla(s).`, 15000)
            } else {
              this.aiModalOpen = true
              this.aiPendingReview = false
            }
          } catch (err) {
            this.showToast('IA: ' + (err.code || err.message))
          } finally {
            this.aiAnalyzing = false
            this.aiAutoAnalyzePending = false
          }
        },

        async aiApply() {
          if (!this.aiModalEdited || !this.aiModalEdited.length) return
          if (!this.validEditedTables()) {
            this.showToast('Revisá los slugs. Deben ser minúsculas y empezar con letra.')
            return
          }
          this.aiApplying = true
          try {
            if (!this.aiSchema || !this.aiSchema.analysisId) {
              this.showToast('No hay análisis de IA cargado.')
              return
            }
            // htmlbox-spec-ai-apply-schema.md: apply hace el trabajo real —
            // crea tablas, inserta TODAS las filas reales del HTML, reescribe
            // el HTML, guarda nueva versión. Por eso mandamos el HTML actual
            // del editor (no el que tenía el server cuando se hizo el
            // análisis): si el tenant editó el box, queremos que el apply
            // corra sobre lo que está viendo AHORA. editorHtml directo.
            const result = await apiFetch(`${CONTROL_PLANE}/api/ai/analyses/${this.aiSchema.analysisId}/apply`, {
              method: 'POST',
              body: { html: this.editorHtml || '' },
            })
            const appliedCount = (result.tables || []).length
            const skippedCount = (result.skipped || []).length
            this.showToast(`Aplicado: ${appliedCount} tabla(s)${skippedCount ? `, ${skippedCount} skipped` : ''} (v${result.newVersion}).`)
            this.closeAiModal()

            // El HTML cambió del lado del servidor (se reemplazaron las
            // declaraciones `const X = [...]` por `await HTMLBox.table(...)` y
            // el <script> ahora tiene type="module"). Refrescamos el editor
            // con el HTML NUEVO y reseleccionamos el box para que preview
            // muestre el cambio.
            await this.loadActiveHtml()
            if (this.currentBox && result.newVersion) {
              // Actualizar la versión local para que coincida con el server.
              this.currentBox = { ...this.currentBox, htmlbox_version: result.newVersion }
            }
            // Refrescar dataTab si está abierta (las tablas nuevas aparecen ahí).
            if (this.activeTab === 'data' && this.$refs.dataTabRef && typeof this.$refs.dataTabRef.loadTables === 'function') {
              await this.$refs.dataTabRef.loadTables()
            }
          } catch (err) {
            this.showToast('Aplicar: ' + (err.code || err.message))
          } finally {
            this.aiApplying = false
          }
        },

        async toggleAutoAnalyze(evt) {
          if (!this.currentBox) return
          const newValue = !!(evt && evt.target && evt.target.checked)
          if (typeof this.currentBox.auto_analyze_on_save === 'boolean') {
            this.currentBox.auto_analyze_on_save = newValue
          } else {
            this.currentBox = { ...this.currentBox, auto_analyze_on_save: newValue }
          }
          this.updatingAutoAnalyze = true
          try {
            await apiFetch(`${CONTROL_PLANE}/api/boxes/${this.currentBox.id}`, {
              method: 'PATCH',
              body: { auto_analyze_on_save: newValue },
            })
          } catch (err) {
            this.toast = 'Auto-IA: ' + (err.code || err.message)
          } finally {
            this.updatingAutoAnalyze = false
          }
        },

        async aiLoadHistory() {
          if (!this.currentBox) return
          try {
            const data = await apiFetch(`${CONTROL_PLANE}/api/ai/analyses?boxId=${this.currentBox.id}`)
            this.aiHistory = data.analyses || []
          } catch (err) {
            debugLog('aiLoadHistory:', err)
          }
        },

        aiReviewPending() {
          this.aiPendingReview = false
          if (this.aiSchema && this.aiModalEdited) {
            this.aiModalOpen = true
          }
        },

        closeAiModal() {
          this.aiModalOpen = false
          this.aiSchema = null
          this.aiModalEdited = null
          this.aiPendingReview = false
        },

        isValidSlug(slug) {
          return /^[a-z][a-z0-9_]{0,40}$/.test(slug || '')
        },

        validEditedTables() {
          if (!this.aiModalEdited || this.aiModalEdited.length === 0) return false
          return this.aiModalEdited.every((t) => this.isValidSlug(t.slug))
        },

        addSampleRow(t) {
          if (!t) return
          if (!Array.isArray(t.sample_rows)) t.sample_rows = []
          if (!Array.isArray(t.columns)) t.columns = []
          const row = {}
          for (const c of t.columns) row[c.name || ''] = ''
          t.sample_rows.push(row)
        },

        async copyToClipboard(s) {
          const text = String(s || '')
          try {
            await navigator.clipboard.writeText(text)
            this.showToast('Copiado al portapapeles.')
          } catch {
            const ta = document.createElement('textarea')
            ta.value = text
            ta.style.position = 'fixed'
            ta.style.opacity = '0'
            document.body.appendChild(ta)
            ta.select()
            document.execCommand('copy')
            document.body.removeChild(ta)
            this.showToast('Copiado.')
          }
        },
      }
    }

    function teamTab() {
      return {

        newEmail: '',
        newDisplayName: '',
        addingUser: false,
        teamStatus: '',
        teamStatusOk: false,

        grantTarget: null,
        grantForm: { scope_type: 'box', scope_id: '' },
        granting: false,

          // Cargar al toque — el tab es tenant-level, no necesita currentBox.
          await this.loadTeam()
          if (!this.grantForm.scope_id) {
            // Pre-seleccionar el primer box si existe, para que el form ya
            // tenga un scope_id válido al abrir el modal.
            const firstBox = (this.$root.boxes || [])[0]
            if (firstBox) this.grantForm.scope_id = firstBox.id
          }

          this.loadingTeam = true
          try {
            const d = await apiFetch('/api/tenant-app-users')
            this.users = d.users || []
          } catch (err) {
            debugLog('teamTab.loadTeam:', err)
            this.users = []
            this.flash('Error cargando equipo: ' + (err.code || err.message), false)
          }
          this.loadingTeam = false
        },

        async addUser() {
          const email = (this.newEmail || '').trim().toLowerCase()
          if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) {
            this.flash('Email inválido.', false)
            return
          }
          this.addingUser = true
          try {
            await apiFetch('/api/tenant-app-users', {
              method: 'POST',
              body: { email, display_name: this.newDisplayName.trim() || null },
            })
            this.newEmail = ''
            this.newDisplayName = ''
            this.showAddForm = false
            await this.loadTeam()
            this.flash('Persona agregada al equipo.', true)
          } catch (err) {
            this.flash('Error: ' + (err.code || err.message), false)
          } finally {
            this.addingUser = false
          }
        },

        async deleteUser(u) {
          if (!confirm(`¿Borrar a ${u.email} del equipo? Esto también quita todos sus accesos y cierra sus sesiones.`)) return
          try {
            await apiFetch(`/api/tenant-app-users/${u.id}`, { method: 'DELETE' })
            await this.loadTeam()
            this.flash('Persona borrada del equipo.', true)
          } catch (err) {
            this.flash('Error: ' + (err.code || err.message), false)
          }
        },

        openGrant(u) {
          this.grantTarget = u
          this.grantModalOpen = true
          // Default: box, pre-seleccionar el primero
          this.grantForm = { scope_type: 'box', scope_id: this.$root.boxes?.[0]?.id || '' }
        },

          if (!this.grantTarget) return
          const body = {
            scope_type: this.grantForm.scope_type,
          }
          if (this.grantForm.scope_type !== 'tenant') {
            if (!this.grantForm.scope_id) {
              this.flash('Elegí un box o workspace.', false)
              return
            }
            body.scope_id = this.grantForm.scope_id
          }
          this.granting = true
          try {
            await apiFetch(`/api/tenant-app-users/${this.grantTarget.id}/access`, {
              method: 'POST',
              body,
            })
            this.grantModalOpen = false
            await this.loadTeam()
            this.flash('Acceso otorgado.', true)
          } catch (err) {
            this.flash('Error: ' + (err.code || err.message), false)
          } finally {
            this.granting = false
          }
        },

        async revokeAccess(u, a) {
          if (!confirm(`¿Revocar el acceso "${a.scope_type}" a esta persona?`)) return
          try {
            await apiFetch(`/api/tenant-app-users/${u.id}/access/${a.id}`, { method: 'DELETE' })
            await this.loadTeam()
            this.flash('Acceso revocado.', true)
          } catch (err) {
            this.flash('Error: ' + (err.code || err.message), false)
          }
        },

        // Etiqueta legible para el scope (muestra box/workspace name, no id)
        scopeLabel(a) {
          if (a.scope_type === 'tenant') return '(todo el tenant)'
          if (a.scope_type === 'workspace') {
            const ws = (this.$root.workspaces || []).find(w => w.id === a.scope_id)
            return ws ? ws.name : a.scope_id
          }
          if (a.scope_type === 'box') {
            const box = (this.$root.boxes || []).find(b => b.id === a.scope_id)
            return box ? `${box.name} (${box.slug})` : a.scope_id
          }
          return a.scope_id || ''
        },

        flash(msg, ok) {
          this.teamStatus = msg
          this.teamStatusOk = !!ok
          setTimeout(() => { this.teamStatus = '' }, 3000)
        },
      }
    }

          } finally {
            this.updatingSignupMode = false
          }
        },

        async addUser() {
          if (!this.$root.currentBox) return
          const email = (this.newEmail || '').trim().toLowerCase()
          if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) {
            this.flash('Email inválido.', false)
            return
          }
          this.addingUser = true
          try {
            await apiFetch(`${RUNTIME}/api/app-auth/${this.$root.currentBox.id}/admin/users`, {
              method: 'POST',
              body: { email, display_name: this.newDisplayName.trim() || null },
            })
            this.newEmail = ''
            this.newDisplayName = ''
            this.showAddForm = false
            await this.loadUsers()
            this.flash('Usuario agregado.', true)
          } catch (err) {
            this.flash('Error: ' + (err.code || err.message), false)
          } finally {
            this.addingUser = false
          }
        },

        async setUserDisabled(userId, disabled) {
          if (!this.$root.currentBox) return
          try {
            await apiFetch(
              `${RUNTIME}/api/app-auth/${this.$root.currentBox.id}/admin/users/${userId}/${disabled ? 'disable' : 'enable'}`,
              { method: 'POST' },
            )
            await this.loadUsers()
            this.flash(disabled ? 'Usuario deshabilitado.' : 'Usuario reactivado.', true)
          } catch (err) {
            this.flash('Error: ' + (err.code || err.message), false)
          }
        },

        async deleteUser(u) {
          if (!this.$root.currentBox) return
          if (!confirm(`¿Borrar a ${u.email}? Esto también cierra sus sesiones activas.`)) return
          try {
            await apiFetch(
              `${RUNTIME}/api/app-auth/${this.$root.currentBox.id}/admin/users/${u.id}`,
              { method: 'DELETE' },
            )
            await this.loadUsers()
            this.flash('Usuario borrado.', true)
          } catch (err) {
            this.flash('Error: ' + (err.code || err.message), false)
          }
        },

        flash(msg, ok) {
          this.usersStatus = msg
          this.usersStatusOk = !!ok
          setTimeout(() => { this.usersStatus = '' }, 3000)
        },
      }
    }

    // ============ Tab Datos (fase 3 v1) ============
    function dataTab() {
      const RUNTIME = (typeof window !== 'undefined' && window.HTMLBOX_RUNTIME_ORIGIN) || ''
      return {
        tables: [],
        loadingTables: false,
        showNewTable: false,
        newTableSlug: '',
        newTableName: '',
        newTableScope: 'private',
        creatingTable: false,
        activeTable: null,
        uploadStrategy: 'replace',
        uploadStatus: null,
        uploadOk: false,
        previewSlug: null,
        previewRows: [],
        previewColumns: [],
        previewLoading: false,

        async init() {
          if (!this.$root.currentBox) return
          await this.loadTables()
        },

        async loadTables() {
          if (!this.$root.currentBox) {
            this.tables = []
            return
          }
          this.loadingTables = true
          try {
            const d = await apiFetch(`${RUNTIME}/api/data/${this.$root.currentBox.id}/tables`)
            this.tables = d.tables || []
          } catch (err) {
            debugLog('dataTab.loadTables:', err)
            this.tables = []
          }
          this.loadingTables = false
        },

        async createTable() {
          const slug = this.newTableSlug.trim().toLowerCase().replace(/[^a-z0-9_]/g, '_')
          if (!slug || !/^[a-z][a-z0-9_]{0,40}$/.test(slug)) {
            alert('Slug inválido. Use minúsculas y guiones bajos, empezando con letra.')
            return
          }
          this.creatingTable = true
          try {
            // Crea metadata (incluye scope — fase 2). El upload posterior
            // respetará el scope si la tabla ya existe.
            await apiFetch(`${RUNTIME}/api/data/${this.$root.currentBox.id}/tables/${slug}`, {
              method: 'POST',
              body: {
                name: this.newTableName.trim() || slug,
                scope: this.newTableScope,
                columns: [],
              },
            })
            this.newTableSlug = ''
            this.newTableName = ''
            this.newTableScope = 'private'
            this.showNewTable = false
            await this.loadTables()
          } catch (err) {
            alert(`Error: ${err.code || err.message}`)
            debugLog('dataTab.createTable:', err)
          }
          this.creatingTable = false
        },

        async toggleTableScope(t) {
          const current = t.scope || 'private'
          const next = current === 'private' ? 'shared' : 'private'
          const msg = next === 'shared'
            ? `Vas a cambiar "${t.slug}" a SHARED. Todos los customers van a poder escribir sobre las mismas filas. ¿Continuar?`
            : `Vas a cambiar "${t.slug}" a PRIVATE. Las filas existentes sin owner_user_id van a quedar inaccesibles para los customers. ¿Continuar?`
          if (!confirm(msg)) return
          try {
            await apiFetch(`${RUNTIME}/api/data/${this.$root.currentBox.id}/tables/${t.slug}/scope`, {
              method: 'POST',
              body: { scope: next },
            })
            await this.loadTables()
          } catch (err) {
            alert(`Error: ${err.code || err.message}`)
            debugLog('dataTab.toggleTableScope:', err)
          }
        },

        selectTable(t) {
          this.activeTable = t.slug
          this.uploadStatus = null
        },

        handleDrop(e) {
          const file = e.dataTransfer.files?.[0]
          if (file) this.uploadFile(file)
        },
        handleFile(e) {
          const file = e.target.files?.[0]
          if (file) this.uploadFile(file)
        },

        async uploadFile(file) {
          if (!this.activeTable) return
          this.uploadStatus = 'Subiendo…'
          this.uploadOk = false
          try {
            const ext = file.name.split('.').pop().toLowerCase()
            let data
            if (ext === 'json') {
              const text = await file.text()
              const rows = JSON.parse(text)
              if (!Array.isArray(rows)) throw new Error('JSON debe ser un array de objetos')
              data = await apiFetch(`${RUNTIME}/api/data/${this.$root.currentBox.id}/tables/${this.activeTable}/upload?strategy=${this.uploadStrategy}`, {
                method: 'POST',
                body: { rows, filename: file.name },
              })
            } else {
              // CSV / TXT — el body es texto crudo, hay que override del Content-Type
              // (apiFetch default es application/json). El spread de headers en
              // apiFetch hace que 'Content-Type: text/csv' gane.
              const text = await file.text()
              data = await apiFetch(`${RUNTIME}/api/data/${this.$root.currentBox.id}/tables/${this.activeTable}/upload?strategy=${this.uploadStrategy}`, {
                method: 'POST',
                headers: { 'Content-Type': 'text/csv' },
                body: text,
              })
            }
            this.uploadStatus = `OK — ${data.inserted} filas insertadas (${data.strategy})`
            this.uploadOk = true
            await this.loadTables()
            this.previewTable({ slug: this.activeTable })
          } catch (err) {
            this.uploadStatus = `Error: ${err.code || err.message}`
            this.uploadOk = false
          }
        },

        async previewTable(t) {
          if (this.previewSlug === t.slug) {
            this.previewSlug = null
            this.previewRows = []
            this.previewColumns = []
            return
          }
          this.previewSlug = t.slug
          this.previewLoading = true
          this.previewRows = []
          this.previewColumns = []
          try {
            const d = await apiFetch(`${RUNTIME}/api/data/${this.$root.currentBox.id}/tables/${t.slug}/rows?limit=10`)
            this.previewRows = d.rows || []
            if (this.previewRows.length > 0) {
              const skip = new Set(['id', 'created_at', 'updated_at'])
              const cols = new Set()
              for (const row of this.previewRows) {
                for (const k of Object.keys(row)) {
                  if (!skip.has(k)) cols.add(k)
                }
              }
              this.previewColumns = Array.from(cols)
            }
          } catch (err) {
            debugLog('dataTab.previewTable:', err)
          }
          this.previewLoading = false
        },
      }
    }
  </script>
